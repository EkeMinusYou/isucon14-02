// characters. Load MySQL's actual weights once rather than approximating folding.

func (s *rideState) loadCollation(ctx context.Context) error {
 var incompatible int
 if err := s.database.GetContext(ctx, &incompatible, `SELECT COUNT(*) FROM information_schema.columns
  WHERE table_schema = DATABASE() AND table_name IN ('users','owners','chairs','coupons','payment_tokens')
  AND collation_name IS NOT NULL AND collation_name <> 'utf8mb4_general_ci'`); err != nil { return err }

 if s.collationWeights != nil { return nil }
 var chars strings.Builder
 for r := rune(0); r <= utf8.MaxRune; r++ {
  if r >= 0xd800 && r <= 0xdfff { continue }
  chars.WriteRune(r)
 }
 var weights []byte
 if err := s.database.GetContext(ctx, &weights,
  "SELECT WEIGHT_STRING(CONVERT(? USING utf8mb4) COLLATE utf8mb4_general_ci)", chars.String()); err != nil { return err }
 const validRunes = utf8.MaxRune + 1 - 0x800
 if len(weights) != validRunes*2 { return fmt.Errorf("unexpected general_ci weights: %d", len(weights)) }
 s.collationWeights = make([]uint16, utf8.MaxRune+1)
 offset := 0

  s.collationWeights[r] = binary.BigEndian.Uint16(weights[offset:])
  offset += 2
 }
 return nil
}

  key = append(key, byte(weight>>8), byte(weight))

}

package main

import "sort"

// Coupons remain in the canonical map after use; the count includes consumed
// invitation benefits, while the small ordered index only contains unused ones.
func (s *rideState) addCoupon(c Coupon) {
 key := s.accountKey(c.Code)
 byCode := s.coupons[c.UserID]
 if byCode == nil { byCode = map[string]Coupon{}; s.coupons[c.UserID] = byCode }
 old, exists := byCode[key]
 if !exists { s.couponCounts[key]++ }
 byCode[key] = c
 queue := s.unusedCoupons[c.UserID]
 if exists && old.UsedBy == nil {
  for i, candidate := range queue {
   if candidate == key { queue = append(queue[:i], queue[i+1:]...); break }
  }
 }
 if c.UsedBy == nil {
  i := sort.Search(len(queue), func(i int) bool {
   other := byCode[queue[i]]
   return other.CreatedAt.After(c.CreatedAt) || (other.CreatedAt.Equal(c.CreatedAt) && queue[i] >= key)
  })
  queue = append(queue, "")
  copy(queue[i+1:], queue[i:])
  queue[i] = key
 }
 s.unusedCoupons[c.UserID] = queue
}

// Estimates preserve the original campaign priority; ride creation requests it
// only for the first ride. Neither selection nor estimates consume a coupon.
func (s *rideState) nextCoupon(userID string, campaignPriority bool) (Coupon, bool) {

  if c, ok := s.coupons[userID][s.accountKey("CP_NEW2024")]; ok && c.UsedBy == nil { return c, true }
 }
 queue := s.unusedCoupons[userID]
 if len(queue) == 0 { return Coupon{}, false }
 return s.coupons[userID][queue[0]], true
}
